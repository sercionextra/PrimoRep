 * Target: Arduino Nano ESP32 (ESP32-S3, 240MHz, FPU 64-bit)
 * 
 * Esecuzione su Core 1: Cinematica Inversa v.4.0 + Generazione Impulsi AccelStepper
 * Esecuzione su Core 0: WebSockets Server per Digital Twin a 60 FPS / 10 Hz Telemetria
 * =========================================================================================
 */

#include <Arduino.h>
#include <WiFi.h>
#include <WebSocketsServer.h>
#include <AccelStepper.h>
#include <math.h>

// =========================================================================
// 1. PINOUT ARDUINO NANO ESP32
// =========================================================================
#define PIN_STEP_1    D2
#define PIN_DIR_1     D3
#define PIN_STEP_2    D4
#define PIN_DIR_2     D5
#define PIN_STEP_3    D6
#define PIN_DIR_3     D7
#define PIN_ENABLE    D8
#define PIN_BAT_ADC   A0

// =========================================================================
// 2. COSTANTI CINEMATICHE E MECCANICHE (Da CORE_MATH v.4.0)
// =========================================================================
const double R_A = 0.30;                  // Raggio pulegge fisse (m)
const double r_b = 0.15;                  // Raggio triangolo specchio (m)
const double H_NOMINAL = 0.35;            // Quota Z nominale (m)
const double STEPS_PER_REV = 3200.0;      // 200 passi * 16 microstep
const double R_CYLINDER = 0.0125;         // Raggio tamburo cabestano (m)
const double STEPS_PER_METER = STEPS_PER_REV / (2.0 * M_PI * R_CYLINDER); // ~40743.6 passi/m
const double MAX_SPEED_M_S = 0.015;       // 15 mm/s
const double MAX_ACCEL_M_S2 = 0.080;      // 80 mm/s^2

// Istanze AccelStepper (Driver Interface = 1)
AccelStepper stepper1(AccelStepper::DRIVER, PIN_STEP_1, PIN_DIR_1);
AccelStepper stepper2(AccelStepper::DRIVER, PIN_STEP_2, PIN_DIR_2);
AccelStepper stepper3(AccelStepper::DRIVER, PIN_STEP_3, PIN_DIR_3);

// Server WebSockets su porta 81 per Digital Twin
WebSocketsServer webSocket = WebSocketsServer(81);

// Variabili di Stato Condivise tra i Core (Thread-Safe)
volatile double targetLengths[3] = {0.3808, 0.3808, 0.3808};
volatile double currentLengths[3] = {0.3808, 0.3808, 0.3808};
volatile double currentTensions[3] = {3.27, 3.27, 3.27};
volatile double batteryVoltage = 12.8;
volatile bool isTrackingActive = true;

// =========================================================================
// 3. CORE MATH: CINEMATICA INVERSA COMPLETA v.4.0 (A 64-BIT NATIVI)
// =========================================================================
void solveInverseKinematics(double nx, double ny, double nz, double outLengths[3]) {
    // Normalizzazione vettore normale
    double norm = sqrt(nx*nx + ny*ny + nz*nz);
    if (norm < 1e-9) { nx = 0; ny = 0; nz = 1; norm = 1; }
    nx /= norm; ny /= norm; nz /= norm;

    // Z-Adattativa C^infinito (Gate 5)
    double tiltDeg = acos(fmax(-1.0, fmin(1.0, nz))) * (180.0 / M_PI);
    double curZ = -H_NOMINAL;
    if (tiltDeg > 28.0) {
        double factor = fmin(1.0, (tiltDeg - 28.0) / 22.0);
        curZ = -H_NOMINAL - (factor * factor) * (0.42 - H_NOMINAL);
    }

    // Punti base fisse A_i (90°, 210°, 330°)
    const double A_pts[3][2] = {
        { 0.0, R_A },
        { -R_A * 0.8660254, -R_A * 0.5 },
        {  R_A * 0.8660254, -R_A * 0.5 }
    };

    // Punti specchio b_i
    const double b_pts[3][2] = {
        { 0.0, r_b },
        { -r_b * 0.8660254, -r_b * 0.5 },
        {  r_b * 0.8660254, -r_b * 0.5 }
    };

    // Matrice Möller-Hughes semplificata (innesco psi = 0)
    double factor = nz > -0.99999 ? 1.0 / (1.0 + nz) : 0.0;
    double R[3][3] = {
        { 1.0 - nx*nx*factor, -nx*ny*factor,       nx },
        { -nx*ny*factor,      1.0 - ny*ny*factor,  ny },
        { -nx,                -ny,                 nz }
    };

    // Calcolo lunghezze per ciascun cavo
    for (int i = 0; i < 3; i++) {
        // px = xG + R*b
        double px = R[0][0]*b_pts[i][0] + R[0][1]*b_pts[i][1];
        double py = R[1][0]*b_pts[i][0] + R[1][1]*b_pts[i][1];
        double pz = curZ + (R[2][0]*b_pts[i][0] + R[2][1]*b_pts[i][1]);

        double dx = A_pts[i][0] - px;
        double dy = A_pts[i][1] - py;
        double dz = 0.0 - pz;

        outLengths[i] = sqrt(dx*dx + dy*dy + dz*dz);
    }
}

// =========================================================================
// 4. TASK CORE 0: NETWORKING & WEBSOCKET TELEMETRY
// =========================================================================
void TaskCore0(void *pvParameters) {
    uint32_t lastTelemetryTime = 0;

    for (;;) {
        webSocket.loop();

        // Trasmissione telemetria verso il browser a 10 Hz (ogni 100 ms)
        if (millis() - lastTelemetryTime >= 100) {
            lastTelemetryTime = millis();

            // Lettura Tensione Batteria via Partitore su A0
            int rawADC = analogRead(PIN_BAT_ADC);
            // Partitore 100k / 22k: Rapporto = (100+22)/22 = 5.545
            batteryVoltage = (rawADC / 4095.0) * 3.3 * 5.545;

            // Formattazione JSON Telemetria Zero-Heap
            char jsonBuffer[256];
            snprintf(jsonBuffer, sizeof(jsonBuffer),
                "{\"l\":[%.1f,%.1f,%.1f],\"t\":[%.2f,%.2f,%.2f],\"bat\":%.1f,\"grip\":1}",
                currentLengths[0]*1000.0, currentLengths[1]*1000.0, currentLengths[2]*1000.0,
                currentTensions[0], currentTensions[1], currentTensions[2],
                batteryVoltage
            );
            webSocket.broadcastTXT(jsonBuffer);
        }
        vTaskDelay(pdMS_TO_TICKS(5));
    }
}

// Ricezione Comandi da Web (Jog Pad o Inseguimento)
void webSocketEvent(uint8_t num, WStype_t type, uint8_t * payload, size_t length) {
    if (type == WStype_TEXT) {
        // Parsing veloce del comando JSON
        // Esempio payload: {"cmd":"target","nx":0.1,"ny":0.2,"nz":0.97}
        double nx, ny, nz;
        if (sscanf((char*)payload, "{\"cmd\":\"target\",\"nx\":%lf,\"ny\":%lf,\"nz\":%lf}", &nx, &ny, &nz) == 3) {
            double solvedL[3];
            solveInverseKinematics(nx, ny, nz, solvedL);
            targetLengths[0] = solvedL[0];
            targetLengths[1] = solvedL[1];
            targetLengths[2] = solvedL[2];
        }
    }
}

// =========================================================================
// 5. SETUP E LOOP PRINCIPALE (CORE 1: REAL-TIME MOTION CONTROL)
// =========================================================================
void setup() {
    Serial.begin(115200);

    pinMode(PIN_ENABLE, OUTPUT);
    digitalWrite(PIN_ENABLE, LOW); // Abilita i driver TMC2209 (Active LOW)

    // Configurazione cinematica dei 3 motori con rampe realistiche
    const double maxStepsSec = MAX_SPEED_M_S * STEPS_PER_METER;      // ~612 passi/s
    const double maxAccelSec2 = MAX_ACCEL_M_S2 * STEPS_PER_METER;    // ~3250 passi/s^2

    AccelStepper* steppers[3] = {&stepper1, &stepper2, &stepper3};
    for (int i = 0; i < 3; i++) {
        steppers[i]->setMaxSpeed(maxStepsSec);
        steppers[i]->setAcceleration(maxAccelSec2);
        steppers[i]->setCurrentPosition(0);
    }

    // Connessione Wi-Fi come Access Point (senza bisogno di router)
    WiFi.softAP("ELIOSTATO_CDPR", "solare123");
    webSocket.begin();
    webSocket.onEvent(webSocketEvent);

    // Avvio del Task di Rete sul Core 0 di ESP32-S3
    xTaskCreatePinnedToCore(
        TaskCore0,        // Funzione del task
        "TaskNetCore0",   // Nome task
        4096,             // Stack size (byte)
        NULL,             // Parametri
        1,                // Priorità
        NULL,             // Task handle
        0                 // Core 0
    );
}

void loop() {
    // Esecuzione sul CORE 1 con massima reattività
    // Conversione lunghezze metri -> passi motore
    long targetSteps1 = (long)((targetLengths[0] - 0.3808) * STEPS_PER_METER);
    long targetSteps2 = (long)((targetLengths[1] - 0.3808) * STEPS_PER_METER);
    long targetSteps3 = (long)((targetLengths[2] - 0.3808) * STEPS_PER_METER);

    stepper1.moveTo(targetSteps1);
    stepper2.moveTo(targetSteps2);
    stepper3.moveTo(targetSteps3);

    // Esecuzione continua del treno d'impulso interpolato
    stepper1.run();
    stepper2.run();
    stepper3.run();

    // Aggiornamento coordinate per telemetria
    currentLengths[0] = 0.3808 + (stepper1.currentPosition() / STEPS_PER_METER);
    currentLengths[1] = 0.3808 + (stepper2.currentPosition() / STEPS_PER_METER);
    currentLengths[2] = 0.3808 + (stepper3.currentPosition() / STEPS_PER_METER);
}
